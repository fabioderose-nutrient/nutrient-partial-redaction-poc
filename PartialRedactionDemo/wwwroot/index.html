One option on the existing SmartRedaction call

using GdPicture14;

using var pdf = new GdPicturePDF();
pdf.LoadFromFile("statement.pdf");

var options = new GdPicturePDF.SmartRedactionOptions
{
    RedactCreditCardNumbers = true,
    PartialRedaction = true,   // keep the last 4 digits
};

pdf.SmartRedaction(options);
pdf.SaveToFile("statement-redacted.pdf");
